3. localStorage is a browser storage API used to store data permanently in key–value pairs.

Data stays even after page refresh

Stores only string values

4.Because localStorage only stores strings.

Solution:

Convert object → JSON string before storing

Convert JSON → object while reading

  import { useState } from "react";

const Register = () => {
  const [user, setUser] = useState({
    username: "",
    password: ""
  });

  const handleChange = (e) => {
    setUser({ ...user, [e.target.name]: e.target.value });
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    localStorage.setItem("user", JSON.stringify(user));
    alert("User Registered");
  };

  return (
    <form onSubmit={handleSubmit}>
      <input name="username" onChange={handleChange} placeholder="Username" />
      <input name="password" type="password" onChange={handleChange} placeholder="Password" />
      <button type="submit">Register</button>
    </form>
  );
};

export default Register;

12.
  import { useState } from "react";

const Login = () => {
  const [login, setLogin] = useState({ username: "", password: "" });
  const [message, setMessage] = useState("");

  const handleSubmit = (e) => {
    e.preventDefault();
    const storedUser = JSON.parse(localStorage.getItem("user"));

    if (
      storedUser &&
      login.username === storedUser.username &&
      login.password === storedUser.password
    ) {
      setMessage("Login Successful");
      localStorage.setItem("isLoggedIn", "true");
    } else {
      setMessage("Invalid Credentials");
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <input onChange={(e) => setLogin({ ...login, username: e.target.value })} placeholder="Username" />
      <input type="password" onChange={(e) => setLogin({ ...login, password: e.target.value })} placeholder="Password" />
      <button>Login</button>
      <p>{message}</p>
    </form>
  );
};

export default Login;

14.
  import { useState, useEffect } from "react";

const EditDeleteUsers = () => {
  const [users, setUsers] = useState([]);

  useEffect(() => {
    setUsers(JSON.parse(localStorage.getItem("users")) || []);
  }, []);

  const deleteUser = (index) => {
    const updated = users.filter((_, i) => i !== index);
    setUsers(updated);
    localStorage.setItem("users", JSON.stringify(updated));
  };

  const editUser = (index) => {
    const updatedName = prompt("Enter new username");
    const updated = users.map((u, i) =>
      i === index ? { ...u, username: updatedName } : u
    );
    setUsers(updated);
    localStorage.setItem("users", JSON.stringify(updated));
  };

  return (
    <div>
      {users.map((user, index) => (
        <div key={index}>
          {user.username}
          <button onClick={() => editUser(index)}>Edit</button>
          <button onClick={() => deleteUser(index)}>Delete</button>
        </div>
      ))}
    </div>
  );
};

export default EditDeleteUsers;
